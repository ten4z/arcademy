
//instalar sdkmanager:
sdkmanager "platform-tools" "platforms;android-36" "build-tools;36.0.0" "emulator"

// instalar cordova no node.js
npm install -g cordova

// criar projeto cordova
cordova create Arcademy com.forjaiapps.arcademy Arcademy

// adicionar a plataforma
cordova platform add ios
cordova platform add android

// Criar a aplicação
cordova build
cordova build ios

// passar parametros gradle
cordova build android -- --gradleArg=-PcdvMinSdkVersion=20

// criar versão release mas em apk
cordova build android --release -- --packageType=apk

// release apk para usuarios do Windows 
cordova build android --release "--" --packageType=apk

// criar a chave da loja
keytool -genkey -v -keystore keystore.jks -keyalg RSA -keysize 2048 -validity 365000 -alias arcademy

// alinhar e assinar app
jarsigner -verbose -sigalg SHA1withRSA -digestalg SHA1 -keystore keystore.jks -signedjar signed-arcademy.aab arcademy.aab arcademy
